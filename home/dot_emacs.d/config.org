* Package management (straight.el)

`straight.el` is bootstrapped in `init.el`, and `(setq straight-use-package-by-default t)` makes `use-package` entries tracked by straight unless overridden.

Notes about lock tracking:
- Packages marked with `:straight nil` are intentionally not tracked by straight and use the built-in Emacs version.
- `straight/versions/default.el` records repository pins, so names there may differ from `use-package` names (for example `which-key` comes from repo `emacs-which-key`).

(defun ik/rename-current-buffer-file (new-name)
  "Rename the file visited by the current buffer to NEW-NAME."
  (interactive
   (let ((filename (buffer-file-name)))
     (unless filename
       (user-error "Current buffer is not visiting a file"))
     (list (read-file-name "Rename file to: " filename))))
  (let ((old-name (buffer-file-name)))
    (if (vc-backend old-name)
        (vc-rename-file old-name new-name)
      (rename-file old-name new-name 1)
      (set-visited-file-name new-name t t))))

(defun ik/copy-current-file-path (&optional style)
  "Copy information about the current file to the kill ring.
STYLE may be `path', `line', `line-column', `directory', `name',
`name-sans-extension', or `project-relative'."
  (interactive)
  (let* ((filename (or (buffer-file-name)
                       (user-error "Current buffer is not visiting a file")))
         (text (pcase (or style 'path)
                 ('line (format "%s:%d" filename (line-number-at-pos)))
                 ('line-column (format "%s:%d:%d" filename
                                       (line-number-at-pos)
                                       (current-column)))
                 ('directory (file-name-directory filename))
                 ('name (file-name-nondirectory filename))
                 ('name-sans-extension
                  (file-name-sans-extension (file-name-nondirectory filename)))
                 ('project-relative
                  (let* ((project (or (project-current)
                                      (user-error "Current buffer is not in a project")))
                         (root (project-root project)))
                    (file-relative-name filename root)))
                 (_ filename))))
    (kill-new text)
    (message "%s" text)))

(defvar ik/killed-file-buffers nil
  "Recently killed file buffers, newest first.")

(defun ik/remember-killed-file-buffer ()
  "Remember the file visited by the current buffer before it is killed."
  (when-let* ((filename (buffer-file-name)))
    (push filename ik/killed-file-buffers)))

(add-hook 'kill-buffer-hook #'ik/remember-killed-file-buffer)

(defun ik/reopen-killed-file-buffer ()
  "Reopen the most recently killed file buffer."
  (interactive)
  (while (and ik/killed-file-buffers
              (not (file-exists-p (car ik/killed-file-buffers))))
    (pop ik/killed-file-buffers))
  (if-let* ((filename (pop ik/killed-file-buffers)))
      (find-file filename)
    (user-error "No killed file buffer to reopen")))

(defun ik/kill-current-buffer-and-window ()
  "Kill the current buffer and delete its window."
  (interactive)
  (let ((window (selected-window)))
    (kill-current-buffer)
    (when (and (window-live-p window) (not (one-window-p t)))
      (delete-window window))))

(defun ik/kill-other-buffers ()
  "Kill all other file-visiting buffers."
  (interactive)
  (let ((current (current-buffer)))
    (mapc (lambda (buffer)
            (when (and (not (eq buffer current))
                       (buffer-file-name buffer))
              (kill-buffer buffer)))
          (buffer-list))))

(defun ik/kill-buffers-matching-regexp (regexp)
  "Kill buffers whose names match REGEXP."
  (interactive "sKill buffers matching regexp: ")
  (mapc (lambda (buffer)
          (when (string-match-p regexp (buffer-name buffer))
            (kill-buffer buffer)))
        (buffer-list)))

(defun ik/kill-other-buffers-and-windows ()

(defun ik/delete-current-file (&optional no-confirm)
  "Delete the current file and kill its buffer.
With NO-CONFIRM, do not ask for confirmation."
  (interactive "P")
  (let ((filename (or (buffer-file-name)
                      (user-error "Current buffer is not visiting a file"))))
    (when (or no-confirm (yes-or-no-p (format "Delete %s? " filename)))
      (delete-file filename)
      (kill-current-buffer)
      (message "Deleted %s" filename))))

    (find-alternate-file (concat "/sudo::" filename))))

(defun ik/open-current-file-externally ()
  "Open the current file with the system default application."
  (interactive)
  (let ((filename (or (buffer-file-name)
                      (user-error "Current buffer is not visiting a file"))))
    (cond
     ((eq system-type 'darwin) (call-process "open" nil 0 nil filename))
     ((eq system-type 'windows-nt) (w32-shell-execute "open" filename))
     (t (call-process "xdg-open" nil 0 nil filename)))))

(defun ik/open-junk-file (extension)
  "Open a timestamped junk file under ~/junk with EXTENSION."
  (interactive "sExtension, without dot: ")
  (let* ((dir (expand-file-name "~/junk/"))
         (filename (expand-file-name
                    (format-time-string
                     (concat "%Y-%m-%d-%H%M%S"
                             (unless (string= extension "") ".")
                             extension))
                    dir)))
    (make-directory dir t)
    (find-file filename)))

(defun ik/toggle-file-tree-sidebar ()
  "Toggle a Dired side window rooted at `default-directory'."
  (interactive)
  (if (window-live-p ik/file-tree-window)
      (progn
        (delete-window ik/file-tree-window)

           '((side . left) (slot . -1) (window-width . 0.25))))))

    (let ((input (read-string "Limit to extensions (comma/space-separated, empty for all): ")))
      (unless (string-empty-p input)
        (mapcar (lambda (extension)
                  (string-remove-prefix "." extension))
                (split-string input "[,[:space:]]+" t))))))

(defun ik/search--extension-args (extensions)
  "Return ripgrep glob arguments limited to EXTENSIONS."
  (mapcan (lambda (extension) (list "-g" (concat "*." extension)))
          extensions))

(defun ik/search--command (pattern targets extensions)
  "Build a ripgrep command for PATTERN, TARGETS, and EXTENSIONS."
  (mapconcat #'shell-quote-argument
             (append (list (ik/search--executable)
                           "--vimgrep"
                           "--smart-case"
                           "--no-heading"
                           "--color=never")
                     (ik/search--extension-args extensions)
                     (list "-e" pattern)
                     targets)
             " "))

(defun ik/search--run (scope pattern targets &optional directory extensions)
  "Run ripgrep for PATTERN over TARGETS in DIRECTORY and remember the result buffer.
SCOPE is a human-readable scope name used for the result buffer."
  (unless targets
    (user-error "No targets for search"))
  (let* ((default-directory (file-name-as-directory (or directory default-directory)))
         (command (ik/search--command pattern targets extensions))
         (buffer-name (format "*rg %s: %s*" scope pattern)))
    (setq ik/search-last-buffer
          (compilation-start command 'grep-mode (lambda (_) buffer-name)))))

        (bounds (bounds-of-thing-at-point 'word)))
    (unless (and num bounds)
      (user-error "No number at point"))
    (delete-region (car bounds) (cdr bounds))
    (while (string-match "\\(.*[0-9]\\)\\([0-9][0-9][0-9].*\\)" num)
      (setq num
            (concat
             (match-string 1 num)
             op
             (match-string 2 num))))
    (insert (format "%s" num))))

(defalias 'ik-add-number-grouping #'ik/add-number-grouping)
