  "Switch to the *scratch* buffer, creating it when necessary."
  (interactive)

(defun ik/revert-buffer-no-confirm ()
  "Revert the current buffer without asking for confirmation."
  (interactive)
  (revert-buffer :ignore-auto :noconfirm))

(defun ik/rename-current-buffer-file (new-name)
  "Rename the file visited by the current buffer to NEW-NAME."
  (interactive
   (let ((filename (buffer-file-name)))
     (unless filename
       (user-error "Current buffer is not visiting a file"))
     (list (read-file-name "Rename file to: " filename))))
  (let ((old-name (buffer-file-name)))
    (if (vc-backend old-name)
        (vc-rename-file old-name new-name)
      (rename-file old-name new-name 1)
      (set-visited-file-name new-name t t))))

(defun ik/copy-current-file-path ()
  "Copy the current buffer's file path to the kill ring."
  (interactive)
  (let ((filename (or (buffer-file-name)
                      (user-error "Current buffer is not visiting a file"))))
    (kill-new filename)
    (message "%s" filename)))

(defun ik/apply-fonts (&optional frame)
  "Apply preferred fonts to FRAME, when they are installed."
  (when (display-graphic-p frame)
    (with-selected-frame (or frame (selected-frame))
      (when (find-font (font-spec :family "Inconsolata"))
        (set-face-attribute 'default nil
                            :family "Inconsolata"
                            :height 140
                            :weight 'normal
                            :width 'normal)
        (set-face-attribute 'fixed-pitch nil
                            :family "Inconsolata"
                            :height 140))
      (let ((variable-pitch-family
             (if (eq system-type 'darwin)
                 "Linux Biolinum"
               "Linux Biolinum O")))
        (when (find-font (font-spec :family variable-pitch-family))
          (set-face-attribute 'variable-pitch nil
                              :family variable-pitch-family
                              :height (if (eq system-type 'darwin) 180 140)))))))

(ik/apply-fonts)
(add-hook 'after-make-frame-functions #'ik/apply-fonts)
(add-hook 'text-mode-hook #'variable-pitch-mode)

#+begin_src emacs-lisp
(defun ik/python-language-server-command ()
  "Return the preferred available Python language server command."
  (cond
   ((executable-find "basedpyright-langserver")
    '("basedpyright-langserver" "--stdio"))
   ((executable-find "pyright-langserver")
    '("pyright-langserver" "--stdio"))
   ((executable-find "pylsp")
    '("pylsp"))))

(defun ik/python-eglot-ensure ()
  "Start Eglot for Python when a language server is installed."
  (interactive)
  (if (ik/python-language-server-command)
      (eglot-ensure)
    (message "No Python language server found; install basedpyright or pyright for Eglot.")))

(use-package eglot
  :straight nil
  :config
  (let ((server-command (ik/python-language-server-command)))
    (when server-command
      (add-to-list 'eglot-server-programs
                   `((python-mode python-ts-mode) . ,server-command)))))

(use-package flymake
  :straight nil)

(defun ik/python-toggle-bool ()
  "Toggle the Python boolean at point between True and False."
  (interactive)
  (let* ((bounds (bounds-of-thing-at-point 'word))
         (bool (and bounds
                    (buffer-substring-no-properties (car bounds) (cdr bounds)))))
    (unless (member bool '("True" "False"))
      (user-error "Point is not on a Python boolean"))
    (delete-region (car bounds) (cdr bounds))
    (insert (if (string= bool "True") "False" "True"))))

(defun ik/python-copy-function-name ()
  "Copy the name of the enclosing Python function to the kill ring."
  (interactive)
  (let ((defun-name (python-info-current-defun)))
    (unless defun-name
      (user-error "Point is not inside a Python function"))
    (let ((function-name (car (last (split-string defun-name "\\.")))))
      (kill-new function-name)
      (message "%s" function-name))))

(use-package python
  :straight nil
  :hook
  ((python-mode python-ts-mode) . ik/python-eglot-ensure)
  :config
  (ik/local-leader
    :keymaps '(python-mode-map python-ts-mode-map)
    "o b" '(ik/python-toggle-bool :which-key "toggle bool")
    "o c" '(ik/python-copy-function-name :which-key "copy function name")
    "n u" '(python-nav-backward-up-list :which-key "up list")
    "n f a" '(python-nav-forward-defun :which-key "next defun")
    "n f e" '(python-nav-backward-defun :which-key "previous defun")
    "n f m" '(python-nav-if-name-main :which-key "if __main__")))
